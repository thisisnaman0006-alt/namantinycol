package main import ( "html/template" "log" "net/http" ) // Dynamic Data structure to send into HTML type PageData struct { Title string Heading string Message string } // Complete HTML Template String const indexHTML = ` {{.Title}}

{{.Heading}}

{{.Message}}

` func homeHandler(w http.ResponseWriter, r *http.Request) { // Parse HTML Template tmpl, err := template.New("index").Parse(indexHTML) if err != nil { http.Error(w, "Error loading template", http.StatusInternalServerError) return } // Data to inject into HTML data := PageData{ Title: "My Go Web App", Heading: "Welcome to Go Lang Web Server", Message: "Ye HTML response Go backend se render ho kar aa raha hai!", } // Render HTML to Browser tmpl.Execute(w, data) } func main() { http.HandleFunc("/", homeHandler) log.Println("Server running at http://localhost:8080") if err := http.ListenAndServe(":8080", nil); err != nil { log.Fatal(err) } }
